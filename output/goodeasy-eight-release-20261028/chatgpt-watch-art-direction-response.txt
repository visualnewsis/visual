반환값의 group을 기존 릴 루트에 붙이고, 필요하면 rotorShell, spoolShell, handleShell만 기존 동작 그룹에 재부착하면 됩니다.

JavaScript
// buildReelExterior(T, materials)
// - T: THREE namespace
// - materials: optional overrides
// - 목적: 기존 동작 로직을 건드리지 않고 외형만 고급형 스피닝 릴로 재구성
// - 좌표 기준은 사용자 제공값 유지
//
// 사용 예:
// const reelSkin = buildReelExterior(THREE, {});
// reelRoot.add(reelSkin.group);
// 혹은 기존 rotor/spool/handle 동작 그룹에 reelSkin.parts.rotorShell 등만 부착

export function buildReelExterior(T, materials = {}) {
  const group = new T.Group();
  group.name = 'reelExterior';

  // ---------------------------------------------------------------------------
  // Anchors from existing mechanism
  // ---------------------------------------------------------------------------
  const spoolCenter = new T.Vector3(1.85, 1.8, 0);
  const bodyDriveCenter = new T.Vector3(-0.4, 1.8, 0);
  const lineRollerRotorLocal = new T.Vector3(1.25, 0.15, -1.02);
  const stemFootY = 3.42;

  // ---------------------------------------------------------------------------
  // Materials
  // ---------------------------------------------------------------------------
  const M = {
    graphite: materials.graphite || new T.MeshStandardMaterial({
      color: 0x23282d,
      metalness: 0.30,
      roughness: 0.52
    }),
    graphiteDark: materials.graphiteDark || new T.MeshStandardMaterial({
      color: 0x171b1f,
      metalness: 0.18,
      roughness: 0.62
    }),
    machined: materials.machined || new T.MeshStandardMaterial({
      color: 0xbfc6cb,
      metalness: 0.88,
      roughness: 0.28
    }),
    machinedDark: materials.machinedDark || new T.MeshStandardMaterial({
      color: 0x8f979d,
      metalness: 0.82,
      roughness: 0.36
    }),
    gold: materials.gold || new T.MeshStandardMaterial({
      color: 0x9d7a2f,
      metalness: 0.92,
      roughness: 0.30
    }),
    rubber: materials.rubber || new T.MeshStandardMaterial({
      color: 0x121416,
      metalness: 0.05,
      roughness: 0.88
    }),
    line: materials.line || new T.MeshStandardMaterial({
      color: 0xd9dbc9,
      metalness: 0.02,
      roughness: 0.92
    })
  };

  // ---------------------------------------------------------------------------
  // Helpers
  // ---------------------------------------------------------------------------
  const deg = T.MathUtils.degToRad;

  function latheX(points, material, segments = 48) {
    // Build Lathe around Y, then rotate so axis runs along X
    const g = new T.LatheGeometry(points, segments);
    g.rotateZ(Math.PI / 2);
    const m = new T.Mesh(g, material);
    m.castShadow = true;
    m.receiveShadow = true;
    return m;
  }

  function capsuleLike(length = 1, r = 0.18, material = M.rubber, radial = 24) {
    const g = new T.CapsuleGeometry(r, Math.max(0.001, length - 2 * r), 8, radial);
    const m = new T.Mesh(g, material);
    m.castShadow = true;
    m.receiveShadow = true;
    return m;
  }

  function extrudeShape(shape, depth, material, bevel = 0.02) {
    const g = new T.ExtrudeGeometry(shape, {
      depth,
      bevelEnabled: true,
      bevelSegments: 3,
      steps: 1,
      bevelSize: bevel,
      bevelThickness: bevel,
      curveSegments: 32
    });
    g.center();
    const m = new T.Mesh(g, material);
    m.castShadow = true;
    m.receiveShadow = true;
    return m;
  }

  function roundedRectShape(w, h, r) {
    const s = new T.Shape();
    const x = -w / 2, y = -h / 2;
    s.moveTo(x + r, y);
    s.lineTo(x + w - r, y);
    s.quadraticCurveTo(x + w, y, x + w, y + r);
    s.lineTo(x + w, y + h - r);
    s.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
    s.lineTo(x + r, y + h);
    s.quadraticCurveTo(x, y + h, x, y + h - r);
    s.lineTo(x, y + r);
    s.quadraticCurveTo(x, y, x + r, y);
    return s;
  }

  function addBoltRing(parent, radius, count, mat, size = 0.03, yOffset = 0, xOffset = 0, zOffset = 0) {
    for (let i = 0; i < count; i++) {
      const a = (i / count) * Math.PI * 2;
      const b = new T.Mesh(new T.CylinderGeometry(size, size, 0.02, 16), mat);
      b.rotation.z = Math.PI / 2;
      b.position.set(
        xOffset,
        yOffset + Math.cos(a) * radius,
        zOffset + Math.sin(a) * radius
      );
      b.castShadow = true;
      parent.add(b);
    }
  }

  // ---------------------------------------------------------------------------
  // BODY : replace blunt circular covers with layered housing
  // ---------------------------------------------------------------------------
  const bodyShell = new T.Group();
  bodyShell.name = 'bodyShell';
  bodyShell.position.copy(bodyDriveCenter);

  // Main body lathe profile (axis = X)
  // Front is flatter; rear swells slightly. No pure disk silhouette.
  const bodyPts = [
    new T.Vector2(0.00, 0.00),
    new T.Vector2(0.05, 0.40),
    new T.Vector2(0.10, 0.63),
    new T.Vector2(0.16, 0.78),
    new T.Vector2(0.28, 0.86),
    new T.Vector2(0.44, 0.84),
    new T.Vector2(0.62, 0.74),
    new T.Vector2(0.82, 0.58),
    new T.Vector2(0.92, 0.44),
    new T.Vector2(0.96, 0.30),
    new T.Vector2(0.98, 0.18),
    new T.Vector2(1.00, 0.00),
  ];
  const bodyCore = latheX(bodyPts, M.graphite, 64);
  bodyCore.rotation.x = deg(2);
  bodyShell.add(bodyCore);

  // Front cover ring - machined highlight, slimmer than old flat disc
  const frontRingPts = [
    new T.Vector2(0.00, 0.00),
    new T.Vector2(0.02, 0.58),
    new T.Vector2(0.05, 0.72),
    new T.Vector2(0.10, 0.80),
    new T.Vector2(0.14, 0.80),
    new T.Vector2(0.16, 0.70),
    new T.Vector2(0.18, 0.55),
    new T.Vector2(0.20, 0.00)
  ];
  const frontRing = latheX(frontRingPts, M.machinedDark, 64);
  frontRing.position.x = -0.10;
  bodyShell.add(frontRing);

  // Rear cover smaller, recessed
  const rearCapPts = [
    new T.Vector2(0.00, 0.00),
    new T.Vector2(0.02, 0.22),
    new T.Vector2(0.05, 0.42),
    new T.Vector2(0.10, 0.58),
    new T.Vector2(0.16, 0.64),
    new T.Vector2(0.22, 0.62),
    new T.Vector2(0.26, 0.46),
    new T.Vector2(0.28, 0.00),
  ];
  const rearCap = latheX(rearCapPts, M.graphiteDark, 48);
  rearCap.position.x = 0.55;
  bodyShell.add(rearCap);

  // Service screws around front cover
  addBoltRing(bodyShell, 0.62, 4, M.machined, 0.028, 0, -0.09, 0);

  // Stem + foot : replace uniform bar with cast S-curve stem and tapered foot
  const stemShape = new T.Shape();
  stemShape.moveTo(-0.08, -0.05);
  stemShape.bezierCurveTo(0.02, -0.10, 0.18, -0.08, 0.28, -0.01);
  stemShape.bezierCurveTo(0.42, 0.08, 0.50, 0.28, 0.50, 0.52);
  stemShape.bezierCurveTo(0.49, 0.88, 0.35, 1.20, 0.18, 1.48);
  stemShape.bezierCurveTo(0.02, 1.76, -0.05, 1.98, -0.02, 2.20);
  stemShape.bezierCurveTo(0.02, 2.42, 0.16, 2.62, 0.34, 2.78);
  stemShape.bezierCurveTo(0.51, 2.93, 0.73, 3.00, 0.95, 3.02);
  stemShape.lineTo(1.02, 2.80);
  stemShape.bezierCurveTo(0.84, 2.78, 0.68, 2.72, 0.56, 2.61);
  stemShape.bezierCurveTo(0.44, 2.50, 0.36, 2.37, 0.33, 2.20);
  stemShape.bezierCurveTo(0.30, 2.03, 0.37, 1.83, 0.53, 1.56);
  stemShape.bezierCurveTo(0.74, 1.19, 0.88, 0.82, 0.88, 0.45);
  stemShape.bezierCurveTo(0.88, 0.14, 0.78, -0.12, 0.59, -0.28);
  stemShape.bezierCurveTo(0.42, -0.43, 0.18, -0.48, -0.08, -0.44);
  stemShape.lineTo(-0.08, -0.05);

  const stem = extrudeShape(stemShape, 0.22, M.graphite, 0.015);
  stem.rotation.y = Math.PI / 2;
  stem.position.set(-0.05, 0.80, 0);
  bodyShell.add(stem);

  const footShape = new T.Shape();
  footShape.moveTo(-0.95, -0.16);
  footShape.bezierCurveTo(-0.40, -0.26, 0.28, -0.26, 0.95, -0.10);
  footShape.lineTo(1.00, 0.02);
  footShape.bezierCurveTo(0.48, 0.12, -0.28, 0.14, -0.98, 0.05);
  footShape.bezierCurveTo(-1.05, 0.00, -1.04, -0.10, -0.95, -0.16);
  const foot = extrudeShape(footShape, 0.18, M.machinedDark, 0.01);
  foot.rotation.x = Math.PI / 2;
  foot.rotation.z = deg(1.5);
  foot.position.set(0.75, stemFootY - bodyDriveCenter.y, 0.02);
  bodyShell.add(foot);

  // A small collar at rotor/bearing junction
  const collar = new T.Mesh(
    new T.CylinderGeometry(0.26, 0.30, 0.20, 32, 1, false),
    M.machinedDark
  );
  collar.rotation.z = Math.PI / 2;
  collar.position.set(-0.18, 0, 0);
  collar.castShadow = true;
  bodyShell.add(collar);

  group.add(bodyShell);

  // ---------------------------------------------------------------------------
  // ROTOR SHELL : asymmetrical modern rotor with distinct main arm / line roller arm
  // ---------------------------------------------------------------------------
  const rotorShell = new T.Group();
  rotorShell.name = 'rotorShell';

4) 이 코드가 결함에 어떻게 대응하는가
A. “동그란 커버가 둔탁하다” 대응
bodyPts를 이용해 앞은 정돈되고 뒤는 조금 부푼 메인 하우징으로 바꿨습니다.
앞/뒤 캡을 분리해 한 덩어리 원판 느낌을 없앴습니다.
frontRing, rearCap, collar로 층을 나눠 제품다운 구성으로 만듭니다.
B. “스풀 플랜지가 거대한 평판이다” 대응
spoolPts에서
앞 립 apex
line wall
rear shoulder
skirt
를 분리했습니다.
dragCap을 별도 Lathe로 만들어 독립 드랙 캡을 확보했습니다.
linePack + 얕은 groove 반복으로 실 감김을 단순 판 텍스처처럼 보이지 않게 했습니다.
C. “로터/베일이 장난감이다” 대응
rotorMainArm과 bailSupportArm을 완전히 다르게 만들었습니다.
bailWire는 얇고 긴 실제 베일 와이어 인상으로 처리했습니다.
lineRoller는 gold 톤의 작은 핵심 부품으로 구분됩니다.
D. “핸들 노브가 구형이다” 대응
구 대신 CapsuleGeometry 기반 타원 러버 노브로 바꿨습니다.
살짝 휘는 handleCurve를 써서 단순 철사 인상을 줄였습니다.
E. “그림자가 검은 덩어리다” 대응
그림자는 아래 조명 설계에서 강도를 낮추고, 실루엣 윤곽광을 분리해
모델을 먹는 그림자가 아니라 접지감만 주는 그림자로 바꿔야 합니다.

// Three.js ES module, no imports; call with the THREE namespace.
// Existing gear meshes, pivots, tooth counts, contacts, and winding API stay untouched.
// Coordinate convention: X/Z = movement plane, Y = height.
// LatheGeometry Vector2(radius, height), never reversed.

export function buildWatchExterior(T, materials = {}) {
  const group = new T.Group(); group.name = 'watchExterior';
  const make = (name, color, metalness, roughness) =>
    materials[name] || new T.MeshStandardMaterial({color, metalness, roughness, side:T.DoubleSide});
  const M = {
    steel:make('steel',0xa9b1b4,.88,.28), polished:make('polished',0xd3d8d6,.94,.18),
    brushed:make('brushed',0x8e999d,.85,.37), dark:make('dark',0x34383b,.58,.48),
    leather:make('leather',0x332b25,.03,.87), edge:make('edge',0x211b18,.04,.94),
    lining:make('lining',0x987e63,.02,.91), stitch:make('stitch',0xb3a28a,.03,.86),
    gold:make('gold',0xa48143,.88,.31), glass:new T.MeshPhysicalMaterial({color:0xffffff,
      metalness:0,roughness:.07,transmission:.98,thickness:.10,ior:1.48,depthWrite:false})
  };
  const parts={};
  function add(parent, geo, mat, name, x=0,y=0,z=0){
    const m=new T.Mesh(geo,mat);m.name=name;m.position.set(x,y,z);
    m.castShadow=true;m.receiveShadow=true;parent.add(m);return m;
  }
  function lathe(parent,profile,mat,name,segments=96){
    return add(parent,new T.LatheGeometry(profile.map(([radius,height])=>new T.Vector2(radius,height)),segments),mat,name);
  }
  function tube(parent,pts,r,mat,name,segments=48){
    const curve=new T.CatmullRomCurve3(pts.map(p=>new T.Vector3(...p)));
    return add(parent,new T.TubeGeometry(curve,segments,r,8,false),mat,name);
  }
  // A shallow contoured case rather than a straight-sided metal cylinder.
  const caseBody=lathe(group,[
    [2.12,-.48],[2.41,-.45],[2.54,-.36],[2.62,-.20],[2.65,-.10],
    [2.64,.02],[2.59,.16],[2.55,.24],[2.49,.31],[2.40,.34],
    [2.38,.26],[2.40,.12],[2.39,-.18],[2.32,-.29],[2.12,-.30]
  ],M.steel,'case-machined-contour');parts.caseBody=caseBody;

      new T.Vector3(x,-.10,z-sign*.14),

      new T.Vector3(x,-.27,z+sign*.62)

export function buildMovementFinishing(T, anchors = {}) {
  const group=new T.Group();group.name='movementFinishing';
  const center=anchors.mainplateCenter || [-.264,0,.3];
  const R=anchors.mainplateRadius ?? 2.4;
  const G=anchors.gearCenters || [[-1.1,.3],[-.264,.3],[.484,.3],[.880,-.3858921198],[1.254,.2618948822]];
  const Y=anchors.gearY || [.30,.48,.66,.84,1.02];
  const bal=anchors.balanceCenter || [.65,-1.05];
  const finish=anchors.materials || {};
  const mat=(name,color,metalness,roughness)=>finish[name] ||
    new T.MeshStandardMaterial({color,metalness,roughness,side:T.DoubleSide});
  const M={plate:mat('plate',0x9ba6a8,.82,.40),bridge:mat('bridge',0xbec4c3,.88,.29),
    bevel:mat('bevel',0xe2e1d9,.91,.18),well:mat('well',0x5a6567,.75,.53),
    screw:mat('screw',0x334b63,.87,.29),bearing:mat('bearing',0x9a7b53,.82,.34),
    jewel:mat('jewel',0x8f243a,.22,.15),crown:mat('crown',0xb5a27a,.83,.37)};
  const parts={};
  function add(parent,geo,material,name,x=0,y=0,z=0){
    const mesh=new T.Mesh(geo,material);mesh.name=name;mesh.position.set(x,y,z);
    mesh.castShadow=true;mesh.receiveShadow=true;parent.add(mesh);return mesh;
  }
  // Extruded geometry starts in XY; rotate +90 degrees about X:
  // shape X->world X; shape Y->world Z; extrusion +Z->world -Y.
  function plateShape(shape,depth,material,name,topY){
    const geo=new T.ExtrudeGeometry(shape,{depth,steps:1,bevelEnabled:true,
      bevelThickness:.014,bevelSize:.015,bevelSegments:2,curveSegments:24});
    geo.rotateX(Math.PI/2);
    return add(group,geo,material,name,0,topY,0);
  }
  const base=new T.Shape();base.absarc(center[0],center[2],R,0,Math.PI*2,false);
  // True through bores in the plate at bearing centers, rather than pink dots.
  for(let i=0;i<G.length;i++){
    const [x,z]=G[i],hole=new T.Path();hole.absarc(x,z,.066,0,Math.PI*2,true);base.holes.push(hole);
  }
  const balHole=new T.Path();balHole.absarc(bal[0],bal[1],.13,0,Math.PI*2,true);base.holes.push(balHole);
  parts.mainplate=plateShape(base,.105,M.plate,'machined-mainplate-with-bores',.19);
  // Recess indicators on the top face — counterbore stepped metal annuli.
  // True deep pockets require subtraction/plate triangulation; no false claim of carved depth.
  for(let i=0;i<G.length;i++){
    const [x,z]=G[i];
    const rim=add(group,new T.RingGeometry(.074,.135,32),M.bevel,`counterbore-rim-${i}`,x,.206,z);
    rim.rotation.x=-Math.PI/2;
    const dark=add(group,new T.RingGeometry(.058,.074,32),M.well,`bushing-recess-${i}`,x,.211,z);
    dark.rotation.x=-Math.PI/2;
  }
  // Curved, small bridge islands leave moving wheels visible.
  // Their bearing pads are at original axis centers, no axes/teeth are moved.
  function bridgeAt(i){
    const [x,z]=G[i],angle=[-.32,.38,-.25,.52,-.55][i],L=[.44,.47,.43,.37,.35][i];
    const dx=Math.cos(angle),dz=Math.sin(angle),nx=-dz,nz=dx;
    const s=new T.Shape();
    s.moveTo(x+nx*.12-dx*.12,z+nz*.12-dz*.12);
    s.bezierCurveTo(x+nx*.17+dx*.04,z+nz*.17+dz*.04,x+nx*.10+dx*L,z+nz*.10+dz*L,x+dx*L,z+dz*L);
    s.bezierCurveTo(x-nx*.13+dx*L,z-nz*.13+dz*L,x-nx*.17+dx*.04,z-nz*.17+dz*.04,x-nx*.12-dx*.12,z-nz*.12-dz*.12);
    s.quadraticCurveTo(x-dx*.24,z-dz*.24,x+nx*.12-dx*.12,z+nz*.12-dz*.12);
    const hole=new T.Path();hole.absarc(x,z,.061,0,Math.PI*2,true);s.holes.push(hole);
    // Y must be above its actual gear, without requiring any moved gear axes.
    const topY=Y[i]+.15;

    if(!parts.bridges)parts.bridges=[];parts.bridges.push(bridge);
    // A screw near the distant support end, with a physically inset slot.
    const sx=x+dx*(L-.095),sz=z+dz*(L-.095);
    const head=add(group,new T.CylinderGeometry(.060,.060,.023,24),M.screw,`screw-head-${i}`,sx,topY+.020,sz);
    const slot=add(group,new T.BoxGeometry(.078,.005,.011),M.well,`screw-slot-${i}`,sx,topY+.034,sz);
    slot.rotation.y=angle;
    // Gold-toned sleeve / jewel in an actual bridge bore, not oversized pink dot.
    const bushing=add(group,new T.RingGeometry(.060,.087,24),M.bearing,`jewel-setting-${i}`,x,topY+.016,z);
    bushing.rotation.x=-Math.PI/2;
    const j=add(group,new T.RingGeometry(.024,.048,24),M.jewel,`small-jewel-annulus-${i}`,x,topY+.018,z);
    j.rotation.x=-Math.PI/2;

  const bcx=bal[0],bcz=bal[1];

  bs.bezierCurveTo(bcx-.42,bcz-.23,bcx-.47,bcz-.52,bcx-.24,bcz-.64);

  const regulator=add(group,new T.TorusGeometry(.11,.016,8,30),M.bearing,'balance-bearing-setting',bcx,.432,bcz);
  regulator.rotation.x=Math.PI/2;
  // Small screw positions can only be functional after checking solid metal below them.
  // No decorative pseudogears or transmission elements are introduced.
  // Finishing texture: restrained concentric milling on outer, EMPTY ring only;
  // it does not cover the working wheels.
  for(let r=2.16;r<=2.29;r+=.065){
    const ring=add(group,new T.TorusGeometry(r,.004,4,128),M.bevel,`perimeter-machining-${r.toFixed(2)}`,
      center[0],.204,center[2]);ring.rotation.x=Math.PI/2;
  }
  return {group,parts,materials:M,anchors:{mainplateCenter:center,gearCenters:G,gearY:Y,balanceCenter:bal},
    note:'Visual supporting geometry only. Keep existing dynamics, gear pivots, jewel contacts and winding functions.'};
}

① 가장 크게 가짜처럼 보이는 형상·재질·접촉 문제 5개
1. 상판·뒤판이 “판자 한 장”처럼 평평하고 두께감/구조감이 없다

지금은 노란 면 하나를 얹은 느낌이라,
사운드홀의 절단 두께, 판의 실제 두께, 아치/돔, 바인딩과 가장자리 단면이 전혀 읽히지 않습니다.
특히 사운드홀 내부가 그냥 빈 구멍처럼 보여서 교육용 관찰 모델로서도 손해입니다.

왜 가짜처럼 보이나

상판이 평면 한 장이면 광 반응이 너무 단순함
사운드홀 테두리에 실제 절단면이 없어 종이처럼 보임
뒤판도 평면이면 진짜 목재 공명판의 느낌이 사라짐
2. 넥이 “원통 막대”처럼 보여 실제 악기 조형이 무너진다

현재는 곡면 넥 프로파일, 헤드-넥 전이(volute 또는 완만한 전이),
힐(heel) 덩어리, 지판이 넥 위에 얹힌 관계, 지판 테이퍼가 거의 읽히지 않습니다.

왜 가짜처럼 보이나

넥 뒷면은 원통이 아니라 위치별로 두께가 달라지는 C/D 프로파일
넥 힐이 없으면 바디와의 구조적 결합이 없어 보임
지판 폭이 안 벌어지면 장난감 우쿨렐레처럼 보임
3. 브리지·새들·브리지핀·현 고정 관계가 모형처럼 단순하다

현은 기타에서 가장 “진짜/가짜”를 빨리 들키는 부위입니다.
지금은 현이 그냥 떠 있는 선처럼 보이고,
너트 슬롯 → 지판 위 현고 → 새들 브레이크 앵글 → 브리지핀 고정의 연속 구조가 안 읽힙니다.

왜 가짜처럼 보이나

현 끝이 브리지 위에 얹힌 듯하고 “몸통 안에서 핀으로 고정되는 느낌”이 없음
새들 높이와 경사, 현의 꺾임 각이 없어 장력감이 없음
헤드 쪽도 페그 축 감김이 없으면 현이 허공에서 시작하는 느낌이 됨
4. 재질이 “색칠된 플라스틱”에 가깝다

상판·측후판·지판·브리지·금속부가 각각 다른 재료로 읽혀야 하는데,
현재는 알베도 한 장 + 단순 광택처럼 보여 목재/금속/본(bone)의 차이가 약합니다.

왜 가짜처럼 보이나

상판 목재결 방향이 안 맞거나 너무 균질
측후판이 휘어진 목재 리본이 아니라 갈색 덩어리처럼 보임
니켈/스틸 금속이 플라스틱처럼 무광이면 값싼 모형처럼 보임
도장(clearcoat) 층이 없으니 고급 악기 특유의 얕은 반사가 없음
5. 분해 화면에서 “접하고 지지하는 관계”가 안 보인다

현재 분해 뷰는 상판-측판-뒤판이 그냥 층으로 벌어지는 그림에 가깝고,
실제 기타 내부의 핵심인 X 브레이싱, 브리지판, 라이닝(kerfed lining), 넥 블록/테일 블록이 빠져 구조 논리가 없습니다.

왜 가짜처럼 보이나

상판이 무엇에 붙는지 안 보임
뒤판이 무엇으로 지지되는지 안 보임
내부 브레이싱이 없으면 사운드홀 안쪽이 텅 빈 모형처럼 보임
교육용 모델인데 “왜 이렇게 붙어 있는지” 설명력이 떨어짐

import * as THREE from 'three';

const MM_TO_U = 2.97 / 648;
const U = mm => mm * MM_TO_U;

function lerp(a, b, t){ return a + (b - a) * t; }
function clamp(x, a, b){ return Math.max(a, Math.min(b, x)); }

// 간단한 해시 노이즈
function hash2(x, y){
  const s = Math.sin(x * 127.1 + y * 311.7) * 43758.5453123;
  return s - Math.floor(s);
}
function noise2(x, y){
  const ix = Math.floor(x), iy = Math.floor(y);
  const fx = x - ix, fy = y - iy;
  const a = hash2(ix, iy);
  const b = hash2(ix + 1, iy);
  const c = hash2(ix, iy + 1);
  const d = hash2(ix + 1, iy + 1);
  const ux = fx * fx * (3 - 2 * fx);
  const uy = fy * fy * (3 - 2 * fy);
  return lerp(lerp(a, b, ux), lerp(c, d, ux), uy);
}

// bookmatch 옵션 포함 절차적 우드 알베도
function makeWoodCanvas({
  width = 1024,
  height = 1024,
  base = '#d8bc84',
  dark = '#b18956',
  pore = '#7f5f3c',
  grainDir = 'x',   // x / y
  lineDensity = 220,
  waviness = 0.06,
  poreAmount = 0.20,
  bookmatch = false
} = {}){
  const canvas = document.createElement('canvas');
  canvas.width = width;
  canvas.height = height;
  const ctx = canvas.getContext('2d');
  const img = ctx.createImageData(width, height);
  const data = img.data;

  const baseC = new THREE.Color(base);
  const darkC = new THREE.Color(dark);
  const poreC = new THREE.Color(pore);

  for(let y = 0; y < height; y++){
    for(let x = 0; x < width; x++){
      let u = x / (width - 1);
      let v = y / (height - 1);

      // center seam 기준 bookmatch
      if(bookmatch){
        const side = u < 0.5 ? u * 2.0 : (1.0 - u) * 2.0;
        u = side;
      }

      const axis = grainDir === 'x' ? u : v;
      const ortho = grainDir === 'x' ? v : u;

      // 고운 연선 + 장주기 왜곡
      const warp =
        (noise2(axis * 4.0, ortho * 8.0) - 0.5) * waviness +
        (noise2(axis * 15.0, ortho * 2.0) - 0.5) * waviness * 0.35;

      const g = Math.sin((axis + warp) * lineDensity * Math.PI * 2.0) * 0.5 + 0.5;
      const g2 = Math.sin((axis + warp * 0.35) * lineDensity * 0.32 * Math.PI * 2.0) * 0.5 + 0.5;
      const grain = clamp(g * 0.55 + g2 * 0.45, 0, 1);

      const pores = Math.pow(noise2(u * 120, v * 120), 8.0) * poreAmount;
      const broad = noise2(u * 6.0, v * 6.0);

      const c = baseC.clone()
        .lerp(darkC, grain * 0.32 + broad * 0.12)
        .lerp(poreC, pores * 0.55);

      const i = (y * width + x) * 4;
      data[i + 0] = Math.round(c.r * 255);
      data[i + 1] = Math.round(c.g * 255);
      data[i + 2] = Math.round(c.b * 255);
      data[i + 3] = 255;
    }
  }

  ctx.putImageData(img, 0, 0);
  return canvas;
}

function bodyHalfWidth(x){
  // x는 body local 기준 대략 -0.95 ~ 0.92 범위
  // 단순 수식보다 샘플 곡선이 수정/검수에 유리함
  const pts = [
    { x: -0.95, z: 0.00 }, // neck joint side
    { x: -0.78, z: 0.32 },
    { x: -0.55, z: 0.58 }, // upper bout max
    { x: -0.15, z: 0.43 }, // waist
    { x:  0.38, z: 0.76 }, // lower bout swell
    { x:  0.76, z: 0.82 },
    { x:  0.92, z: 0.00 }  // tail
  ];

  // piecewise linear + smoothstep 보간
  for(let i=0; i<pts.length-1; i++){
    const a = pts[i], b = pts[i+1];
    if(x >= a.x && x <= b.x){
      const t = (x - a.x) / (b.x - a.x);
      const s = t * t * (3 - 2 * t);
      return lerp(a.z, b.z, s);
    }
  }
  return 0;
}

function makeBodyOutlineShape(segments = 120){
  const shape = new THREE.Shape();
  let first = true;
  for(let i=0; i<=segments; i++){
    const t = i / segments;
    const x = lerp(-0.95, 0.92, t);
    const z = bodyHalfWidth(x);
    if(first){ shape.moveTo(x, z); first = false; }
    else shape.lineTo(x, z);
  }
  for(let i=segments; i>=0; i--){
    const t = i / segments;
    const x = lerp(-0.95, 0.92, t);
    const z = -bodyHalfWidth(x);
    shape.lineTo(x, z);
  }
  shape.closePath();

  // 사운드홀
  const hole = new THREE.Path();
  hole.absellipse(-0.10, 0.0, 0.205, 0.205, 0, Math.PI * 2, false, 0);
  shape.holes.push(hole);

  return shape;
}

// 플레이트 중앙 돔
function topDome(x, z){
  // lower bout 중심 쪽에 약한 돔
  const cx = 0.18, cz = 0.0;
  const rx = 0.95, rz = 0.85;
  const dx = (x - cx) / rx;
  const dz = (z - cz) / rz;
  const r2 = dx * dx + dz * dz;
  const h = Math.max(0, 1 - r2);
  return U(1.8) * h * h; // 약 1.8mm
}

function backArch(x, z){
  const cx = 0.20, cz = 0.0;
  const rx = 0.95, rz = 0.85;
  const dx = (x - cx) / rx;
  const dz = (z - cz) / rz;
  const r2 = dx * dx + dz * dz;
  const h = Math.max(0, 1 - r2);
  return U(5.0) * h * h; // 약 5mm
}

function buildPlate({
  kind = 'top',   // top / back
  yOuter = 0.101,
  thickness = U(2.8),
  material = MAT.top
} = {}){
  const shape = makeBodyOutlineShape(160);
  const geo = new THREE.ExtrudeGeometry(shape, {
    depth: thickness,
    bevelEnabled: false,
    curveSegments: 64,
    steps: 1
  });

  // ExtrudeGeometry 기본 축 정리
  geo.rotateX(Math.PI / 2);

  // top은 위에서 아래로, back은 아치 방향 반대로
  const pos = geo.attributes.position;
  const v = new THREE.Vector3();

  for(let i = 0; i < pos.count; i++){
    v.fromBufferAttribute(pos, i);

    // 회전 후 좌표계 해석:
    // x = x, y = depth축, z = original y(shape)
    // 편집 편의를 위해 y를 원하는 판 위치로 재설정
    const x = v.x;
    const z = v.z;

    if(kind === 'top'){
      // 윗면을 yOuter 근처에 맞추고 underside는 thickness 만큼 아래
      const onOuter = v.y < thickness * 0.5; // 임시 판정
      const dome = topDome(x, z);
      const localY = onOuter ? yOuter + dome : (yOuter - thickness + dome * 0.70);
      v.y = localY;
    } else {
      // back outer가 더 아치
      const arch = backArch(x, z);
      const edgeY = -0.115;
      const outerY = edgeY - arch;
      const innerY = outerY + thickness;
      const onOuter = v.y < thickness * 0.5;
      v.y = onOuter ? outerY : innerY;
    }

    pos.setXYZ(i, v.x, v.y, v.z);
  }
  geo.computeVertexNormals();

  const mesh = new THREE.Mesh(geo, material);
  return mesh;
}

function buildRim(){
  const outer = [];
  const inner = [];
  const xMin = -0.95, xMax = 0.92;
  const rimThick = U(2.1);

  const yTop = 0.085;
  const yBottom = -0.115;

  const seg = 140;
  for(let i=0; i<=seg; i++){
    const t = i / seg;
    const x = lerp(xMin, xMax, t);
    const zOut = bodyHalfWidth(x);
    const zIn = Math.max(0, zOut - rimThick);
    outer.push(new THREE.Vector3(x, yTop, zOut));
    inner.push(new THREE.Vector3(x, yTop, zIn));
  }

  // 간단 구현: 좌우 별도 strip를 만들어도 되고, 실제론 contour loft가 더 깔끔
  const group = new THREE.Group();

  function makeSide(sign = 1){
    const positions = [];
    for(let i=0; i<seg; i++){
      const a = outer[i], b = outer[i+1];
      const ai = inner[i], bi = inner[i+1];

      const z0 = a.z * sign, z1 = b.z * sign, zi0 = ai.z * sign, zi1 = bi.z * sign;

      const quad = [
        [a.x, yTop, z0], [b.x, yTop, z1], [b.x, yBottom, z1],
        [a.x, yTop, z0], [b.x, yBottom, z1], [a.x, yBottom, z0], // outer wall

        [ai.x, yTop, zi0], [ai.x, yBottom, zi0], [bi.x, yBottom, zi1],
        [ai.x, yTop, zi0], [bi.x, yBottom, zi1], [bi.x, yTop, zi1], // inner wall

        [a.x, yTop, z0], [ai.x, yTop, zi0], [bi.x, yTop, zi1],
        [a.x, yTop, z0], [bi.x, yTop, zi1], [b.x, yTop, z1], // top edge

        [a.x, yBottom, z0], [b.x, yBottom, z1], [bi.x, yBottom, zi1],
        [a.x, yBottom, z0], [bi.x, yBottom, zi1], [ai.x, yBottom, zi0] // bottom edge
      ];
      positions.push(...quad.flat());
    }

    const geo = new THREE.BufferGeometry();
    geo.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
    geo.computeVertexNormals();

    const m = new THREE.Mesh(geo, MAT.sideBack);
    return m;
  }

  group.add(makeSide(1));
  group.add(makeSide(-1));

  // 라이닝: 시각적으로는 strip + notch bump 만으로도 충분
  function makeLining(y, sign, inset = U(1.0)){
    const pts = [];
    for(let i=0; i<=seg; i++){
      const t = i / seg;
      const x = lerp(xMin, xMax, t);
      const z = (bodyHalfWidth(x) - rimThick + inset) * sign;
      pts.push(new THREE.Vector3(x, y, z));
    }
    const curve = new THREE.CatmullRomCurve3(pts);
    const geo = new THREE.TubeGeometry(curve, 180, U(3.5), 8, false);
    return new THREE.Mesh(geo, new THREE.MeshPhysicalMaterial({
      color: '#8a6a4b',
      roughness: 0.62,
      metalness: 0.0
    }));
  }

  group.add(makeLining(0.083, 1));
  group.add(makeLining(0.083, -1));
  group.add(makeLining(-0.113, 1));
  group.add(makeLining(-0.113, -1));

  return group;
}

function buildFretboard(){
  const x0 = -2.34;
  const x1 = -0.66; // body extension 조금 포함 가능
  const nSeg = 80;

  const nutW = U(43);
  const endW = U(56);
  const boardT = U(6.2);
  const radius = U(304.8);

  const positions = [];
  const indices = [];
  const vertsPerRow = 10;

  function yArc(z, halfW){
    // 12" radius 지판 곡률
    const zz = z;
    return Math.sqrt(Math.max(0.000001, radius * radius - zz * zz)) - radius;
  }

  for(let i=0; i<=nSeg; i++){
    const tx = i / nSeg;
    const x = lerp(x0, x1, tx);
    const halfW = lerp(nutW * 0.5, endW * 0.5, tx);

    for(let j=0; j<vertsPerRow; j++){
      const tj = j / (vertsPerRow - 1);
      const z = lerp(-halfW, halfW, tj);
      const yTop = 0.190 - U(6.2) + yArc(z, halfW); // 너트 높이 기준 근사
      const yBottom = yTop - boardT;

      positions.push(x, yTop, z);
      positions.push(x, yBottom, z);
    }
  }

  const rowStride = vertsPerRow * 2;
  for(let i=0; i<nSeg; i++){
    for(let j=0; j<rowStride - 2; j++){
      const a = i * rowStride + j;
      const b = a + rowStride;
      const c = b + 1;
      const d = a + 1;
      indices.push(a, b, d, b, c, d);
    }
  }

  const geo = new THREE.BufferGeometry();
  geo.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
  geo.setIndex(indices);
  geo.computeVertexNormals();

  return new THREE.Mesh(geo, MAT.board);
}

function fretX(n){
  return -2.34 + 2.97 * (1 - Math.pow(2, -n / 12));
}

function boardWidthAt(x){
  const x0 = -2.34, x1 = -0.66;
  const t = clamp((x - x0) / (x1 - x0), 0, 1);
  return lerp(U(43), U(56), t);
}

function buildFrets(group){
  for(let n=1; n<=20; n++){
    const x = fretX(n);
    const w = boardWidthAt(x) - U(1.6); // 끝에서 약간 inset
    const geo = new THREE.CapsuleGeometry(U(0.55), w, 4, 12);
    geo.rotateZ(Math.PI / 2);

    const fret = new THREE.Mesh(geo, MAT.fret);
    fret.position.set(x, 0.190 - U(5.0), 0);
    // 엄밀히는 지판 반경에 맞춰 눕혀야 함
    group.add(fret);
  }
}

function buildNut(){
  const geo = new THREE.BoxGeometry(U(5.5), U(3.6), U(43));
  const nut = new THREE.Mesh(geo, MAT.bone);
  nut.position.set(-2.34, 0.190 - U(1.8), 0);
  return nut;
}

function buildBridgeGroup(){
  const g = new THREE.Group();

  // 브리지 본체
  const bridgeL = U(152);   // Z 방향
  const bridgeW = U(31);    // X 방향
  const bridgeT = U(8.5);   // Y 방향

  const body = new THREE.Mesh(
    new THREE.BoxGeometry(bridgeW, bridgeT, bridgeL),
    MAT.bridge
  );
  body.position.set(0.64, 0.101 + bridgeT * 0.5 - U(0.4), 0);
  g.add(body);

  // 새들
  const saddle = new THREE.Mesh(
    new THREE.BoxGeometry(U(3.0), U(3.2), U(72)),
    MAT.bone
  );
  saddle.position.set(0.63, 0.206 - U(1.6), 0);
  g.add(saddle);

  // 브리지핀
  const pinZ = [-.124, -.074, -.025, .025, .074, .124];
  for(let i=0; i<6; i++){
    const pin = new THREE.Mesh(
      new THREE.CylinderGeometry(U(1.3), U(2.1), U(7.0), 12),
      MAT.bone
    );
    pin.position.set(0.71, 0.101 + U(3.6), pinZ[i]);
    g.add(pin);
  }

  return g;
}

const STRING_Z_NUT = [-.080, -.048, -.016, .016, .048, .080];
const STRING_Z_SAD = [-.124, -.074, -.025, .025, .074, .124];
const STRING_R = [U(0.16), U(0.20), U(0.28), U(0.36), U(0.48), U(0.60)];

function makeStringCurve(i){
  const z0 = STRING_Z_NUT[i];
  const z1 = STRING_Z_SAD[i];

  const pts = [
    new THREE.Vector3(-2.34, 0.190, z0),               // nut
    new THREE.Vector3(-1.50, 0.178, lerp(z0, z1, 0.32)),
    new THREE.Vector3(-0.40, 0.162, lerp(z0, z1, 0.68)),
    new THREE.Vector3( 0.63, 0.206, z1),               // saddle crest
    new THREE.Vector3( 0.71, 0.185, z1)                // pin hole side
  ];
  return new THREE.CatmullRomCurve3(pts);
}

function buildStrings(){
  const g = new THREE.Group();
  for(let i=0; i<6; i++){
    const curve = makeStringCurve(i);
    const geo = new THREE.TubeGeometry(curve, 80, STRING_R[i], 8, false);
    const mat = i < 2 ? MAT.stringPlain : MAT.stringWound;
    const s = new THREE.Mesh(geo, mat);
    s.userData = { stringIndex: i, baseCurve: curve };
    g.add(s);
  }
  return g;
}

function updateStringVibration(stringMesh, t, amp){
  const i = stringMesh.userData.stringIndex;
  const z0 = STRING_Z_NUT[i];
  const z1 = STRING_Z_SAD[i];

  const curvePts = [];
  const N = 24;
  for(let k=0; k<=N; k++){
    const s = k / N;
    const x = lerp(-2.34, 0.63, s);
    const z = lerp(z0, z1, s);

    // 양 끝 고정, 중앙 최대
    const env = Math.sin(Math.PI * s);
    // 1차 진동 모드 + 감쇠
    const y =
      lerp(0.190, 0.206, s) +
      Math.sin(t * 42.0) * amp * env;

    curvePts.push(new THREE.Vector3(x, y, z));
  }

  // saddle 뒤쪽 꺾임 추가
  curvePts.push(new THREE.Vector3(0.71, 0.185, z1));

  const curve = new THREE.CatmullRomCurve3(curvePts);
  stringMesh.geometry.dispose();
  stringMesh.geometry = new THREE.TubeGeometry(curve, 64, STRING_R[i], 8, false);
}

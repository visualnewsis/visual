아래는 기존 구조 좌표는 유지하고 외형만 새로 덮는 기준입니다.

기준 좌표
spoolCenter = (1.85, 1.8, 0)
spool radius ≈ 0.8
spool depth ≈ 0.9
bodyDriveCenter = (-0.4, 1.8, 0)
handleAxis = z
stemFootY = 3.42
lineRoller rotorLocal = (1.25, 0.15, -1.02)
heroEye = [5.8, 3.4, 6]
권장 비례
바디
메인 바디 최대 직경: 1.65 ~ 1.78
전면 커버 두께: 0.34 ~ 0.42
후면 하우징 깊이: 0.78 ~ 0.92
바디는 완전한 원반이 아니라 앞은 납작하고 뒤가 살짝 부풀어야 함
스풀
전체 외경: 기존 radius 0.8 기준 외경 1.54~1.62
립 두께: 0.10 ~ 0.14
립 깊이: 0.14 ~ 0.20
스커트 후방 볼륨: 0.45 ~ 0.58
드랙 캡 직경: 스풀 외경의 약 0.38 ~ 0.46
로터
로터 외곽 회전 반경: 1.10 ~ 1.18
메인 로터 암은 넓고 유선형
라인롤러 쪽 암은 짧고 기능적
좌우가 똑같으면 안 됨
핸들
바디 중심에서 핸들 암 끝까지: 1.25 ~ 1.45
핸들 암 굵기: 0.08 ~ 0.12
노브 크기: 길이 0.42 ~ 0.55, 두께 0.18 ~ 0.24
노브는 세로로 긴 타원형 러버
풋/스템
스템은 단순 기둥 금지
스템은 뒤에서 올라와 바디 상부에 녹아드는 S 커브
풋은 얇은 판이 아니라 전후로 가늘어지는 캐스팅 형상
3) 미감 전용 외형 헬퍼 코드

아래 코드는 동작부를 새로 쓰지 않고, 기존 축/좌표계에 맞춰 외형만 재구성하는 헬퍼입니다.
반환값의 group을 기존 릴 루트에 붙이고, 필요하면 rotorShell, spoolShell, handleShell만 기존 동작 그룹에 재부착하면 됩니다.

JavaScript
// buildReelExterior(T, materials)
// - T: THREE namespace
// - materials: optional overrides
// - 목적: 기존 동작 로직을 건드리지 않고 외형만 고급형 스피닝 릴로 재구성
// - 좌표 기준은 사용자 제공값 유지
//
// 사용 예:
// const reelSkin = buildReelExterior(THREE, {});
// reelRoot.add(reelSkin.group);
// 혹은 기존 rotor/spool/handle 동작 그룹에 reelSkin.parts.rotorShell 등만 부착

export function buildReelExterior(T, materials = {}) {
  const group = new T.Group();
  group.name = 'reelExterior';

  // ---------------------------------------------------------------------------
  // Anchors from existing mechanism
  // ---------------------------------------------------------------------------
  const spoolCenter = new T.Vector3(1.85, 1.8, 0);
  const bodyDriveCenter = new T.Vector3(-0.4, 1.8, 0);
  const lineRollerRotorLocal = new T.Vector3(1.25, 0.15, -1.02);
  const stemFootY = 3.42;

  // ---------------------------------------------------------------------------
  // Materials
  // ---------------------------------------------------------------------------
  const M = {
    graphite: materials.graphite || new T.MeshStandardMaterial({
      color: 0x23282d,
      metalness: 0.30,
      roughness: 0.52
    }),
    graphiteDark: materials.graphiteDark || new T.MeshStandardMaterial({
      color: 0x171b1f,
      metalness: 0.18,
      roughness: 0.62
    }),
    machined: materials.machined || new T.MeshStandardMaterial({
      color: 0xbfc6cb,
      metalness: 0.88,
      roughness: 0.28
    }),
    machinedDark: materials.machinedDark || new T.MeshStandardMaterial({
      color: 0x8f979d,
      metalness: 0.82,
      roughness: 0.36
    }),
    gold: materials.gold || new T.MeshStandardMaterial({
      color: 0x9d7a2f,
      metalness: 0.92,
      roughness: 0.30
    }),
    rubber: materials.rubber || new T.MeshStandardMaterial({
      color: 0x121416,
      metalness: 0.05,
      roughness: 0.88
    }),
    line: materials.line || new T.MeshStandardMaterial({
      color: 0xd9dbc9,
      metalness: 0.02,
      roughness: 0.92
    })
  };

  // ---------------------------------------------------------------------------
  // Helpers
  // ---------------------------------------------------------------------------
  const deg = T.MathUtils.degToRad;

  function latheX(points, material, segments = 48) {
    // Build Lathe around Y, then rotate so axis runs along X
    const g = new T.LatheGeometry(points, segments);
    g.rotateZ(Math.PI / 2);
    const m = new T.Mesh(g, material);
    m.castShadow = true;
    m.receiveShadow = true;
    return m;
  }

  function capsuleLike(length = 1, r = 0.18, material = M.rubber, radial = 24) {
    const g = new T.CapsuleGeometry(r, Math.max(0.001, length - 2 * r), 8, radial);
    const m = new T.Mesh(g, material);
    m.castShadow = true;
    m.receiveShadow = true;
    return m;
  }

  function extrudeShape(shape, depth, material, bevel = 0.02) {
    const g = new T.ExtrudeGeometry(shape, {
      depth,
      bevelEnabled: true,
      bevelSegments: 3,
      steps: 1,
      bevelSize: bevel,
      bevelThickness: bevel,
      curveSegments: 32
    });
    g.center();
    const m = new T.Mesh(g, material);
    m.castShadow = true;
    m.receiveShadow = true;
    return m;
  }

  function roundedRectShape(w, h, r) {
    const s = new T.Shape();
    const x = -w / 2, y = -h / 2;
    s.moveTo(x + r, y);
    s.lineTo(x + w - r, y);
    s.quadraticCurveTo(x + w, y, x + w, y + r);
    s.lineTo(x + w, y + h - r);
    s.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
    s.lineTo(x + r, y + h);
    s.quadraticCurveTo(x, y + h, x, y + h - r);
    s.lineTo(x, y + r);
    s.quadraticCurveTo(x, y, x + r, y);
    return s;
  }

  function addBoltRing(parent, radius, count, mat, size = 0.03, yOffset = 0, xOffset = 0, zOffset = 0) {
    for (let i = 0; i < count; i++) {
      const a = (i / count) * Math.PI * 2;
      const b = new T.Mesh(new T.CylinderGeometry(size, size, 0.02, 16), mat);
      b.rotation.z = Math.PI / 2;
      b.position.set(
        xOffset,
        yOffset + Math.cos(a) * radius,
        zOffset + Math.sin(a) * radius
      );
      b.castShadow = true;
      parent.add(b);
    }
  }

  // ---------------------------------------------------------------------------
  // BODY : replace blunt circular covers with layered housing
  // ---------------------------------------------------------------------------
  const bodyShell = new T.Group();
  bodyShell.name = 'bodyShell';
  bodyShell.position.copy(bodyDriveCenter);

  // Main body lathe profile (axis = X)
  // Front is flatter; rear swells slightly. No pure disk silhouette.
  const bodyPts = [
    new T.Vector2(0.00, 0.00),
    new T.Vector2(0.05, 0.40),
    new T.Vector2(0.10, 0.63),
    new T.Vector2(0.16, 0.78),
    new T.Vector2(0.28, 0.86),
    new T.Vector2(0.44, 0.84),
    new T.Vector2(0.62, 0.74),
    new T.Vector2(0.82, 0.58),
    new T.Vector2(0.92, 0.44),
    new T.Vector2(0.96, 0.30),
    new T.Vector2(0.98, 0.18),
    new T.Vector2(1.00, 0.00),
  ];
  const bodyCore = latheX(bodyPts, M.graphite, 64);
  bodyCore.rotation.x = deg(2);
  bodyShell.add(bodyCore);

  // Front cover ring - machined highlight, slimmer than old flat disc
  const frontRingPts = [
    new T.Vector2(0.00, 0.00),
    new T.Vector2(0.02, 0.58),
    new T.Vector2(0.05, 0.72),
    new T.Vector2(0.10, 0.80),
    new T.Vector2(0.14, 0.80),
    new T.Vector2(0.16, 0.70),
    new T.Vector2(0.18, 0.55),
    new T.Vector2(0.20, 0.00)
  ];
  const frontRing = latheX(frontRingPts, M.machinedDark, 64);
  frontRing.position.x = -0.10;
  bodyShell.add(frontRing);

  // Rear cover smaller, recessed
  const rearCapPts = [
    new T.Vector2(0.00, 0.00),
    new T.Vector2(0.02, 0.22),
    new T.Vector2(0.05, 0.42),
    new T.Vector2(0.10, 0.58),
    new T.Vector2(0.16, 0.64),
    new T.Vector2(0.22, 0.62),
    new T.Vector2(0.26, 0.46),
    new T.Vector2(0.28, 0.00),
  ];
  const rearCap = latheX(rearCapPts, M.graphiteDark, 48);
  rearCap.position.x = 0.55;
  bodyShell.add(rearCap);

  // Service screws around front cover
  addBoltRing(bodyShell, 0.62, 4, M.machined, 0.028, 0, -0.09, 0);

  // Stem + foot : replace uniform bar with cast S-curve stem and tapered foot
  const stemShape = new T.Shape();
  stemShape.moveTo(-0.08, -0.05);
  stemShape.bezierCurveTo(0.02, -0.10, 0.18, -0.08, 0.28, -0.01);
  stemShape.bezierCurveTo(0.42, 0.08, 0.50, 0.28, 0.50, 0.52);
  stemShape.bezierCurveTo(0.49, 0.88, 0.35, 1.20, 0.18, 1.48);
  stemShape.bezierCurveTo(0.02, 1.76, -0.05, 1.98, -0.02, 2.20);
  stemShape.bezierCurveTo(0.02, 2.42, 0.16, 2.62, 0.34, 2.78);
  stemShape.bezierCurveTo(0.51, 2.93, 0.73, 3.00, 0.95, 3.02);
  stemShape.lineTo(1.02, 2.80);
  stemShape.bezierCurveTo(0.84, 2.78, 0.68, 2.72, 0.56, 2.61);
  stemShape.bezierCurveTo(0.44, 2.50, 0.36, 2.37, 0.33, 2.20);
  stemShape.bezierCurveTo(0.30, 2.03, 0.37, 1.83, 0.53, 1.56);
  stemShape.bezierCurveTo(0.74, 1.19, 0.88, 0.82, 0.88, 0.45);
  stemShape.bezierCurveTo(0.88, 0.14, 0.78, -0.12, 0.59, -0.28);
  stemShape.bezierCurveTo(0.42, -0.43, 0.18, -0.48, -0.08, -0.44);
  stemShape.lineTo(-0.08, -0.05);

  const stem = extrudeShape(stemShape, 0.22, M.graphite, 0.015);
  stem.rotation.y = Math.PI / 2;
  stem.position.set(-0.05, 0.80, 0);
  bodyShell.add(stem);

  const footShape = new T.Shape();
  footShape.moveTo(-0.95, -0.16);
  footShape.bezierCurveTo(-0.40, -0.26, 0.28, -0.26, 0.95, -0.10);
  footShape.lineTo(1.00, 0.02);
  footShape.bezierCurveTo(0.48, 0.12, -0.28, 0.14, -0.98, 0.05);
  footShape.bezierCurveTo(-1.05, 0.00, -1.04, -0.10, -0.95, -0.16);
  const foot = extrudeShape(footShape, 0.18, M.machinedDark, 0.01);
  foot.rotation.x = Math.PI / 2;
  foot.rotation.z = deg(1.5);
  foot.position.set(0.75, stemFootY - bodyDriveCenter.y, 0.02);
  bodyShell.add(foot);

  // A small collar at rotor/bearing junction
  const collar = new T.Mesh(
    new T.CylinderGeometry(0.26, 0.30, 0.20, 32, 1, false),
    M.machinedDark
  );
  collar.rotation.z = Math.PI / 2;
  collar.position.set(-0.18, 0, 0);
  collar.castShadow = true;
  bodyShell.add(collar);

  group.add(bodyShell);

  // ---------------------------------------------------------------------------
  // ROTOR SHELL : asymmetrical modern rotor with distinct main arm / line roller arm
  // ---------------------------------------------------------------------------
  const rotorShell = new T.Group();
  rotorShell.name = 'rotorShell';
  rotorShell.position.copy(spoolCenter); // tie visually to spool axis
  // If existing logic already has rotor group, re-parent children as needed.

  // Rotor hub ring
  const rotorHub = new T.Mesh(
    new T.TorusGeometry(0.42, 0.06, 16, 48),
    M.graphiteDark
  );
  rotorHub.rotation.y = Math.PI / 2;
  rotorHub.position.set(-0.48, 0, 0);
  rotorHub.castShadow = true;
  rotorShell.add(rotorHub);

  // Main rotor arm (broad, sculpted, asymmetrical)
  const rotorArmShape = new T.Shape();
  rotorArmShape.moveTo(-0.08, -0.12);
  rotorArmShape.bezierCurveTo(0.12, -0.18, 0.40, -0.14, 0.70, 0.02);
  rotorArmShape.bezierCurveTo(0.96, 0.15, 1.12, 0.28, 1.22, 0.44);
  rotorArmShape.bezierCurveTo(1.05, 0.53, 0.86, 0.60, 0.64, 0.60);
  rotorArmShape.bezierCurveTo(0.38, 0.60, 0.16, 0.52, -0.02, 0.34);
  rotorArmShape.bezierCurveTo(-0.12, 0.22, -0.16, 0.08, -0.08, -0.12);

  const rotorMainArm = extrudeShape(rotorArmShape, 0.11, M.graphite, 0.012);
  rotorMainArm.rotation.y = Math.PI / 2;
  rotorMainArm.position.set(-0.35, 0.05, 0.66);
  rotorMainArm.rotation.x = deg(-5);
  rotorShell.add(rotorMainArm);

  // Opposite bail support arm (slimmer, more technical)
  const bailArmShape = new T.Shape();
  bailArmShape.moveTo(-0.05, -0.08);
  bailArmShape.bezierCurveTo(0.06, -0.12, 0.28, -0.10, 0.50, 0.00);
  bailArmShape.bezierCurveTo(0.66, 0.07, 0.82, 0.18, 0.92, 0.31);
  bailArmShape.bezierCurveTo(0.82, 0.38, 0.68, 0.42, 0.52, 0.43);
  bailArmShape.bezierCurveTo(0.28, 0.42, 0.08, 0.35, -0.04, 0.24);
  bailArmShape.bezierCurveTo(-0.11, 0.15, -0.12, 0.03, -0.05, -0.08);

  const bailSupportArm = extrudeShape(bailArmShape, 0.09, M.machinedDark, 0.010);
  bailSupportArm.rotation.y = Math.PI / 2;
  bailSupportArm.position.set(-0.28, 0.02, -0.73);
  bailSupportArm.rotation.x = deg(4);
  rotorShell.add(bailSupportArm);

  // Bail wire - thinner and cleaner
  const bailCurve = new T.CatmullRomCurve3([
    new T.Vector3(0.20, 0.70, 0.84),
    new T.Vector3(0.54, 0.88, 0.36),
    new T.Vector3(0.72, 0.86, -0.06),
    new T.Vector3(0.54, 0.74, -0.54),
    new T.Vector3(0.22, 0.28, -0.90)
  ]);
  const bailWire = new T.Mesh(
    new T.TubeGeometry(bailCurve, 48, 0.035, 10, false),
    M.machined
  );
  bailWire.castShadow = true;
  rotorShell.add(bailWire);

  // Line roller assembly
  const lineRoller = new T.Group();
  lineRoller.position.copy(lineRollerRotorLocal);

  const roller = new T.Mesh(
    new T.CylinderGeometry(0.050, 0.065, 0.20, 24),
    M.gold
  );
  roller.rotation.z = Math.PI / 2;
  roller.castShadow = true;
  lineRoller.add(roller);

  const rollerFork = new T.Mesh(
    new T.CylinderGeometry(0.018, 0.018, 0.25, 12),
    M.machinedDark
  );
  rollerFork.position.x = -0.04;
  rollerFork.rotation.z = deg(20);
  rollerFork.castShadow = true;
  lineRoller.add(rollerFork);

  rotorShell.add(lineRoller);

  // Rotor nose / visual weight near front
  const rotorNose = new T.Mesh(
    new T.SphereGeometry(0.10, 20, 18),
    M.machinedDark
  );
  rotorNose.scale.set(1.1, 0.8, 0.8);
  rotorNose.position.set(0.20, 0.18, -0.86);
  rotorNose.castShadow = true;
  rotorShell.add(rotorNose);

  group.add(rotorShell);

  // ---------------------------------------------------------------------------
  // SPOOL SHELL : deep lip, proper drum volume, independent drag cap
  // ---------------------------------------------------------------------------
  const spoolShell = new T.Group();
  spoolShell.name = 'spoolShell';
  spoolShell.position.copy(spoolCenter);

  // Main spool profile around X axis
  // Deeper front lip, real inner step, fuller skirt, less "flat plate"
  const spoolPts = [
    new T.Vector2(0.00, 0.00),
    new T.Vector2(0.03, 0.26),
    new T.Vector2(0.08, 0.55),
    new T.Vector2(0.14, 0.73),
    new T.Vector2(0.18, 0.79), // front lip apex
    new T.Vector2(0.26, 0.79),
    new T.Vector2(0.32, 0.72),
    new T.Vector2(0.40, 0.58), // line wall
    new T.Vector2(0.53, 0.55),
    new T.Vector2(0.68, 0.57), // line bed
    new T.Vector2(0.82, 0.62), // rear shoulder
    new T.Vector2(0.95, 0.51), // skirt
    new T.Vector2(1.03, 0.35),
    new T.Vector2(1.08, 0.18),
    new T.Vector2(1.10, 0.00),
  ];
  const spoolBody = latheX(spoolPts, M.machined, 64);
  spoolBody.castShadow = true;
  spoolShell.add(spoolBody);

  // Inner arbor / dark center
  const arbor = new T.Mesh(
    new T.CylinderGeometry(0.20, 0.20, 0.92, 32),
    M.graphiteDark
  );
  arbor.rotation.z = Math.PI / 2;
  arbor.position.set(0.50, 0, 0);
  arbor.castShadow = true;
  spoolShell.add(arbor);

  // Front lip accent ring
  const lipAccent = new T.Mesh(
    new T.TorusGeometry(0.73, 0.018, 10, 48),
    M.gold
  );
  lipAccent.rotation.y = Math.PI / 2;
  lipAccent.position.x = 0.20;
  spoolShell.add(lipAccent);

  // Rear skirt accent
  const rearAccent = new T.Mesh(
    new T.TorusGeometry(0.49, 0.015, 10, 48),
    M.machinedDark
  );
  rearAccent.rotation.y = Math.PI / 2;
  rearAccent.position.x = 0.83;
  spoolShell.add(rearAccent);

  // Independent drag cap
  const dragCapPts = [
    new T.Vector2(0.00, 0.00),
    new T.Vector2(0.04, 0.18),
    new T.Vector2(0.08, 0.28),
    new T.Vector2(0.14, 0.31),
    new T.Vector2(0.20, 0.28),
    new T.Vector2(0.26, 0.20),
    new T.Vector2(0.30, 0.00)
  ];
  const dragCap = latheX(dragCapPts, M.graphite, 48);
  dragCap.position.set(-0.08, 0, 0);
  spoolShell.add(dragCap);

  // Drag cap face insert
  const dragCapFace = new T.Mesh(
    new T.CylinderGeometry(0.18, 0.20, 0.05, 24),
    M.machinedDark
  );
  dragCapFace.rotation.z = Math.PI / 2;
  dragCapFace.position.set(-0.12, 0, 0);
  spoolShell.add(dragCapFace);

  // Knurl suggestion as shallow notches around drag cap edge
  for (let i = 0; i < 18; i++) {
    const a = (i / 18) * Math.PI * 2;
    const notch = new T.Mesh(
      new T.BoxGeometry(0.03, 0.03, 0.012),
      M.graphiteDark
    );
    notch.position.set(-0.07, Math.cos(a) * 0.245, Math.sin(a) * 0.245);
    notch.lookAt(-0.07, 0, 0);
    spoolShell.add(notch);
  }

  // Line pack as slightly inset cylindrical band, not just texture slapped on plate
  const linePack = new T.Mesh(
    new T.CylinderGeometry(0.56, 0.56, 0.32, 48, 1, true),
    M.line
  );
  linePack.rotation.z = Math.PI / 2;
  linePack.position.set(0.54, 0, 0);
  spoolShell.add(linePack);

  // Light circumferential grooves to imply winding layers
  for (let i = 0; i < 14; i++) {
    const groove = new T.Mesh(
      new T.TorusGeometry(0.56, 0.0035, 8, 64),
      M.graphiteDark
    );
    groove.rotation.y = Math.PI / 2;
    groove.position.set(0.43 + i * 0.022, 0, 0);
    groove.scale.set(1, 0.985, 0.985);
    spoolShell.add(groove);
  }

  group.add(spoolShell);

  // ---------------------------------------------------------------------------
  // HANDLE SHELL : curved arm + elliptical rubber knob
  // ---------------------------------------------------------------------------
  const handleShell = new T.Group();
  handleShell.name = 'handleShell';
  handleShell.position.copy(bodyDriveCenter);

  // Handle boss at axis
  const handleBoss = new T.Mesh(
    new T.CylinderGeometry(0.12, 0.12, 0.12, 24),
    M.machined
  );
  handleBoss.castShadow = true;
  handleShell.add(handleBoss);

  // Curved handle arm (cleaner than straight tube)
  const handleCurve = new T.CatmullRomCurve3([
    new T.Vector3(0.00, 0.00, 0.00),
    new T.Vector3(0.26, -0.02, 0.00),
    new T.Vector3(0.50, -0.10, 0.00),
    new T.Vector3(0.74, -0.26, 0.00),
    new T.Vector3(0.92, -0.38, 0.00),
    new T.Vector3(1.04, -0.42, 0.00)
  ]);
  const handleArm = new T.Mesh(
    new T.TubeGeometry(handleCurve, 32, 0.055, 12, false),
    M.machined
  );
  handleArm.castShadow = true;
  handleShell.add(handleArm);

  // Knob stem offset
  const knobStem = new T.Mesh(
    new T.CylinderGeometry(0.030, 0.030, 0.16, 14),
    M.machinedDark
  );
  knobStem.position.set(1.06, -0.43, 0.0);
  knobStem.rotation.z = deg(18);
  handleShell.add(knobStem);

  // Elliptical rubber knob (hero-quality vs sphere)

  knob.rotation.z = deg(90);
  knob.scale.set(1.00, 0.72, 1.18);
  knob.position.set(1.19, -0.48, 0.0);

  const knobCap = new T.Mesh(

  knobCap.rotation.x = Math.PI / 2;
  knobCap.position.set(1.42, -0.48, 0.0);
  handleShell.add(knobCap);

  group.add(handleShell);

  // ---------------------------------------------------------------------------
  // Small finishing pieces for product feel
  // ---------------------------------------------------------------------------
  // Spool shaft collar
  const shaftCollar = new T.Mesh(

    M.machinedDark
  );
  shaftCollar.rotation.z = Math.PI / 2;
  shaftCollar.position.copy(spoolCenter).add(new T.Vector3(-0.36, 0, 0));
  group.add(shaftCollar);

  // Front trim ring near rotor-spool junction
  const trimRing = new T.Mesh(
    new T.TorusGeometry(0.46, 0.018, 12, 48),
    M.gold
  );
  trimRing.rotation.y = Math.PI / 2;
  trimRing.position.copy(spoolCenter).add(new T.Vector3(-0.20, 0, 0));
  group.add(trimRing);

  // ---------------------------------------------------------------------------
  // Return structure
  // ---------------------------------------------------------------------------
  return {

    parts: {

4) 이 코드가 결함에 어떻게 대응하는가
A. “동그란 커버가 둔탁하다” 대응
bodyPts를 이용해 앞은 정돈되고 뒤는 조금 부푼 메인 하우징으로 바꿨습니다.
앞/뒤 캡을 분리해 한 덩어리 원판 느낌을 없앴습니다.
frontRing, rearCap, collar로 층을 나눠 제품다운 구성으로 만듭니다.
B. “스풀 플랜지가 거대한 평판이다” 대응
spoolPts에서
앞 립 apex
line wall
rear shoulder
skirt
를 분리했습니다.
dragCap을 별도 Lathe로 만들어 독립 드랙 캡을 확보했습니다.
linePack + 얕은 groove 반복으로 실 감김을 단순 판 텍스처처럼 보이지 않게 했습니다.
C. “로터/베일이 장난감이다” 대응
rotorMainArm과 bailSupportArm을 완전히 다르게 만들었습니다.
bailWire는 얇고 긴 실제 베일 와이어 인상으로 처리했습니다.
lineRoller는 gold 톤의 작은 핵심 부품으로 구분됩니다.
D. “핸들 노브가 구형이다” 대응
구 대신 CapsuleGeometry 기반 타원 러버 노브로 바꿨습니다.
살짝 휘는 handleCurve를 써서 단순 철사 인상을 줄였습니다.
E. “그림자가 검은 덩어리다” 대응
그림자는 아래 조명 설계에서 강도를 낮추고, 실루엣 윤곽광을 분리해
모델을 먹는 그림자가 아니라 접지감만 주는 그림자로 바꿔야 합니다.
